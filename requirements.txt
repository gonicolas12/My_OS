# My_OS — dépendances Python
# Versions non figées : ajuster au premier `pip install` selon l'environnement Arch.
# Installer aussi Ollama séparément : https://ollama.com/download

# --- Interface (couche 4) ---
PySide6              # popup Qt natif

# --- Daemon / système (couche 1 & pilotage) ---
pynput               # capture du raccourci clavier global (X11)
psutil               # processus et ressources système
dbus-python ; sys_platform == "linux"   # D-Bus (luminosité, audio, réseau) — Linux uniquement (jalon 3)
                     # pas de wheel Windows (compilation native) ; alternative cross-platform : QtDBus de PySide6

# --- Port Wayland (jalon 5, EXPÉRIMENTAL — Linux uniquement) ---
# Raccourci global Wayland via le portal org.freedesktop.portal.GlobalShortcuts :
# nécessite une boucle GLib pour les signaux D-Bus. X11 (pynput) reste le chemin
# nominal ; ces paquets ne sont requis que sous une session Wayland.
PyGObject ; sys_platform == "linux"      # GLib MainLoop pour le portal (import paresseux)
                     # Popup Wayland en surface overlay : plugin SYSTÈME `layer-shell-qt`
                     # (paquet pacman, PAS un paquet pip) + Qt Wayland. Voir docs/INSTALLATION.md.

# --- IPC (daemon <-> popup) ---
pyzmq                # messagerie robuste sur socket (optionnel : module socket standard suffit au départ)

# --- Modèles (couche 3) ---
ollama               # client local Qwen — requis seulement si model.backend=ollama
                     # (import paresseux : le mode stub et les tests s'en passent)
anthropic            # API cloud Claude (mode opt-in)

# --- Secrets ---
keyring              # stockage clé API via le trousseau OS (Secret Service)

# --- Configuration ---
PyYAML               # lecture de config.yaml

# --- Outils dev ---
pytest               # tests (cibler permissions/ en priorité)
ruff                 # lint + format
